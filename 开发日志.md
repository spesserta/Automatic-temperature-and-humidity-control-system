### 一、介绍

采用WPF C# 依托于依赖注入，控制反转库DependencyInjection实现的开发解耦，结合CommunityToolkit.Mvvm、MahApps、LiveCharts、HandyControl、IconPacks等组件共同开发而成。<br>

硬件上以PLC为核心，结合多个传感器实现数据输入。

#### 1、采用技术

##### CommunityToolkit.Mvvm

一个现代快速且模块化的MVVM库，支持属性更改通知(ObserableObject)，命令支持(RelayCommand和AsyncRelayCommand)，消息传递、依赖注入、输入验证、源生成器、跨平台支持等。支持Winform和WPF等



##### LiveChart

做数据可视化的库，可以做各种图标

##### HandyControl

开源的自定义控件，比微软原生控件更好看

##### Sqllite

轻量型数据库，是ORM框架，提供LINQ查询、事务管理等操作。代码优先，数据库优先，有迁移功能，可以异步操作

然后UserRepository类把用户操作的接口实例化：

```c#
using CompanyDatabase;
using CompanyDatabase.Entities;
using System;
using System.Collections.Generic;
using System.Text;

namespace CompanyDatabaseSqllite
{
    //实现用户表的操作
    public class userRepository : IUserRepository
    {
        private SqliteDbContext db = new();
        public bool Delete(UserEntity entity)  //删除
        {
            db.Entry(entity).State = Microsoft.EntityFrameworkCore.EntityState.Deleted;
            return db.SaveChanges() > 0;
        }

        public List<UserEntity> GetAll()  //获取User表所有数据
        {
            return db.Users.ToList();
        }

        public bool Insert(UserEntity entity)  //新增用户
        {
            db.Entry(entity).State = Microsoft.EntityFrameworkCore.EntityState.Added;
            return db.SaveChanges() > 0;
        }

        public bool Login(string username, string password, out UserEntity user)  //登录
        {
            user = null;
            var list = db.Users.Where(t => t.UserName.Equals(username) && t.Password.Equals(password)).ToList();
            if(list!=null && list.Count > 0)
            {
                user = list.FirstOrDefault() ?? new();  //选查找到的第一条，没有就new一个
                return true;
            }
            return false;
        }

        public UserEntity? Search(string username)  //按名查找
        {
            return db.Users.Where(t=>t.UserName == username).FirstOrDefault();
        }
        

        public bool Update(UserEntity entity)  //更新用户
        {
            db.Entry(entity).State = Microsoft.EntityFrameworkCore.EntityState.Modified;
            return db.SaveChanges() > 0;
        }
    }
}
```

```c#
    {
        public static IServiceProvider Load()
        {
            var services = new ServiceCollection();

            services.AddSingleton<ShellView>();
            services.AddSingleton<ShellViewModel>();
            services.AddSingleton<InitlizeView>();
```

```c#
            if (Initialized)    //已经有了就直接返回不重复初始化
                return Initialized;

            plcConfig = config;
            bool result = DoInit();
            if (result)
            {
                Initialized = true;
                Message = "初始化成功";
            }

            return result;
        }

        //在子类中必须实现的加载方法
        public abstract bool DoInit();


        public abstract ushort[] Read(string address, ushort length);

        public abstract bool Write(string address, ushort value);

        public abstract bool Write(string address, ushort[] value);
    }
}
```

```c#
namespace CompanyCore.Logger

    public class Logs
    {
        public static string path = AppDomain.CurrentDomain.BaseDirectory + "logs"; //日志地址

        protected static void WriteLog(string type,string content)
        {
            lock (LogLocker) //lock多线程环境下写入日志时发生冲突
            {
                if(!Directory.Exists(path))  //如果日志目录不存在就创建目录

                }
                string time = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss.fff");

                StreamWriter sw = File.AppendText(filename);
                string write_content = time + " " + type + ":" + content;
                sw.WriteLine(write_content);
                sw.Close();
            }
        }

        public static void Debug(string content) => WriteLog("debug", content);
        public static void Info(string content) => WriteLog("info", content);
        public static void Error(string content) => WriteLog("error", content);
        public static void Error(Exception ex) => WriteLog("error", ex.Message +ex.StackTrace);

    }
```

```c#
namespace CompanyShell.Controls
{
    /// <summary>
    /// Valve.xaml 的交互逻辑
    /// </summary>
    public partial class Valve : UserControl
    {
        public Valve()
        {
            InitializeComponent();
        }

        //依赖属性propbp快速插入
        //当IsOpen被修改时，IsOpenProperty也将被修改，从而影响到IsOpenChangedCallback方法开始执行
        public bool IsOpen
        {
            get { return (bool)GetValue(IsOpenProperty); }
            set { SetValue(IsOpenProperty, value); }
        }
        // Using a DependencyProperty as the backing store for IsOpen.  This enables animation, styling, binding, etc...
        public static readonly DependencyProperty IsOpenProperty =
            DependencyProperty.Register("IsOpen", typeof(bool), typeof(Valve), new PropertyMetadata(false,new PropertyChangedCallback(IsOpenChangedCallback)));

        private static void IsOpenChangedCallback(DependencyObject d, DependencyPropertyChangedEventArgs e)
        {
            //开状态带绿色的竖条，关闭的话竖条消失
            var control = (Valve)d;
            if(control == null) 
            {
                return;
            }
            if(e.NewValue is bool isopen)
            {
                if (isopen)
                {
                    control._open.Visibility = Visibility.Visible;
                    control._close.Visibility = Visibility.Collapsed;
                }
                else
                {
                    control._open.Visibility = Visibility.Collapsed;
                    control._close.Visibility = Visibility.Visible;
                }

                
            }
        }
    }
}
```
