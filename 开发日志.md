### 一、介绍

采用WPF C# 依托于依赖注入，控制反转库DependencyInjection实现的开发解耦，结合CommunityToolkit.Mvvm、MahApps、LiveCharts、HandyControl、IconPacks等组件共同开发而成。<br>

硬件上以PLC为核心，结合多个传感器实现数据输入。

#### 1、采用技术

##### CommunityToolkit.Mvvm

一个现代快速且模块化的MVVM库，支持属性更改通知(ObserableObject)，命令支持(RelayCommand和AsyncRelayCommand)，消息传递、依赖注入、输入验证、源生成器、跨平台支持等。支持Winform和WPF等



##### LiveChart

做数据可视化的库，可以做各种图标

##### HandyControl

开源的自定义控件，比微软原生控件更好看

##### Sqllite

轻量型数据库，是ORM框架，提供LINQ查询、事务管理等操作。代码优先，数据库优先，有迁移功能，可以异步操作

然后UserRepository类把用户操作的接口实例化：

```c#
using CompanyDatabase;
using CompanyDatabase.Entities;
using System;
using System.Collections.Generic;
using System.Text;

namespace CompanyDatabaseSqllite
{
    //实现用户表的操作
    public class userRepository : IUserRepository
    {
        private SqliteDbContext db = new();
        public bool Delete(UserEntity entity)  //删除
        {
            db.Entry(entity).State = Microsoft.EntityFrameworkCore.EntityState.Deleted;
            return db.SaveChanges() > 0;
        }

        public List<UserEntity> GetAll()  //获取User表所有数据
        {
            return db.Users.ToList();
        }

        public bool Insert(UserEntity entity)  //新增用户
        {
            db.Entry(entity).State = Microsoft.EntityFrameworkCore.EntityState.Added;
            return db.SaveChanges() > 0;
        }

        public bool Login(string username, string password, out UserEntity user)  //登录
        {
            user = null;
            var list = db.Users.Where(t => t.UserName.Equals(username) && t.Password.Equals(password)).ToList();
            if(list!=null && list.Count > 0)
            {
                user = list.FirstOrDefault() ?? new();  //选查找到的第一条，没有就new一个
                return true;
            }
            return false;
        }

        public UserEntity? Search(string username)  //按名查找
        {
            return db.Users.Where(t=>t.UserName == username).FirstOrDefault();
        }
        

        public bool Update(UserEntity entity)  //更新用户
        {
            db.Entry(entity).State = Microsoft.EntityFrameworkCore.EntityState.Modified;
            return db.SaveChanges() > 0;
        }
    }
}
```

```c#
            if (Initialized)    //已经有了就直接返回不重复初始化
                return Initialized;

            plcConfig = config;
            bool result = DoInit();
            if (result)
            {
                Initialized = true;
                Message = "初始化成功";
            }

            return result;
        }

        //在子类中必须实现的加载方法
        public abstract bool DoInit();


        public abstract ushort[] Read(string address, ushort length);

        public abstract bool Write(string address, ushort value);

        public abstract bool Write(string address, ushort[] value);
    }
}
```

```c#
        //颜色动画定义
        private ColorAnimation colorAnimation = new ColorAnimation()
        {
            From = Color.FromRgb(219, 219, 219),

            Duration = TimeSpan.FromSeconds(2),
            AutoReverse = true,
            RepeatBehavior = RepeatBehavior.Forever,
        };

        public Heater()
        {
            InitializeComponent();

            // 保存初始背景色
            closeBrush = _border.Background;

            // 创建一个新的 SolidColorBrush 并赋值给 Path 的 Stroke
            // 这样我们就拥有了一个完全由代码控制、未被冻结的画笔对象
            // 注意：这里假设你初始想要灰色，如果XAML里设置了其他颜色，请相应调整 From 的值
            pathStrokeBrush = new SolidColorBrush(Color.FromRgb(219, 219, 219));
            _path.Stroke = pathStrokeBrush;

            set { SetValue(IsOpenProperty, value); }

            DependencyProperty.Register("IsOpen", typeof(bool), typeof(Heater), new PropertyMetadata(false, new PropertyChangedCallback(IsOpenChangedCallback)));

        private static void IsOpenChangedCallback(DependencyObject d, DependencyPropertyChangedEventArgs e)
        {
            var control = (Heater)d;
            if (control == null) return;

            if (e.NewValue is bool isopen)
            {

                {
                    control._border.Background = control.openBrush;
                    control._switch.Text = "ON";

                    // 对我们在构造函数中创建的 pathStrokeBrush 进行动画
                    // 而不是直接对 control._path.Stroke 进行操作
                    control.pathStrokeBrush.BeginAnimation(SolidColorBrush.ColorProperty, control.colorAnimation);

                else   // 按钮关闭

                    control._border.Background = control.closeBrush;
                    control._switch.Text = "OFF";

                    // 停止动画，恢复原状（可选）
                    control.pathStrokeBrush.BeginAnimation(SolidColorBrush.ColorProperty, null);
                    // 如果需要关闭时立即变回灰色，可以手动设置颜色：
                    // control.pathStrokeBrush.Color = Color.FromRgb(219, 219, 219);
                }
            }
        }
```

```c#
using CommunityToolkit.Mvvm.ComponentModel;
using LiveCharts;
using LiveCharts.Defaults;
using System;
using System.Collections.Generic;
using System.Text;

namespace CompanyShell.Model
{
    /// <summary>
    /// 程序文件
    /// </summary>
    public class ProgramEntity : ObservableObject
    {
        public ProgramEntity()
        {
            Step = MaxXAxis / 12;
        }
        public string? Guid { get; set; }
        private string name = string.Empty;
        /// <summary>
        /// 程序名
        /// </summary>
        public string Name
        {
            get { return name; }
            set { SetProperty(ref name, value); }
        }

        private ChartValues<ObservablePoint> temperatures = new();
        /// <summary>
        /// 温度曲线坐标集合
        /// </summary>
        public ChartValues<ObservablePoint> Temperatures
        {
            get { return temperatures; }
            set { SetProperty(ref temperatures, value); }
        }

        private ChartValues<ObservablePoint> humidities = new();
        /// <summary>
        /// 湿度曲线坐标集合
        /// </summary>
        public ChartValues<ObservablePoint> Humidities
        {
            get { return humidities; }
            set { SetProperty(ref humidities, value); }
        }

        private double maxXAxis = 1440;
        /// <summary>
        /// X轴最大值（分钟数）
        /// </summary>
        public double MaxXAxis
        {
            get { return maxXAxis; }
            set { SetProperty(ref maxXAxis, value); }
        }

        private double step = 120;
        /// <summary>
        /// X轴刻度值
        /// </summary>
        public double Step
        {
            get { return step; }
            set { SetProperty(ref step, value); }
        }
        /// <summary>
        /// 以天为单位增加X轴的量程
        /// </summary>
        /// <returns></returns>
        public double IncreaseXAxis()
        {
            MaxXAxis += 1440;
            if (MaxXAxis > 1440 * 7)
            {
                MaxXAxis = 1440 * 7;
            }
            Step = MaxXAxis / 12;
            return MaxXAxis;
        }

        /// <summary>
        /// 以天为单位减少X轴的量程
        /// </summary>
        /// <returns></returns>
        public double DecreaseXAxis()
        {
            MaxXAxis -= 1440;
            if (MaxXAxis < 1440)
            {
                MaxXAxis = 1440;
            }
            Step = MaxXAxis / 12;
            return MaxXAxis;
        }
    }
}
```
